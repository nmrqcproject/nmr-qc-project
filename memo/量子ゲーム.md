## 量子ゲーム（cardputer）
```c++
#include <M5Cardputer.h>
#include <WiFi.h>
#include <math.h>

// 裏画面（スプライト）の用意。チカチカ防止用
M5Canvas canvas(&M5Cardputer.Display);

// 1. 量子状態の変数 (2-Qubit)
float r00 = 1.0, i00 = 0.0;
float r01 = 0.0, i01 = 0.0;
float r10 = 0.0, i10 = 0.0;
float r11 = 0.0, i11 = 0.0;

// 2. 環境ノイズ・ゲーム変数
int wifi_count = 0;
float base_noise = 0.0;
float dynamic_noise = 0.0;
float calib_val = 0.5;
float current_error = 0.0;
float fidelity = 100.0;

unsigned long coherence_score = 0; // スコアシステム
bool calib_mode = false;

// 位相エラー注入（EFNMRでの磁場不均一による位相のズレを再現）
void applyPhaseError() {
    if (current_error < 0.02) return;
    float phase = current_error * PI; 
    float cos_p = cos(phase); float sin_p = sin(phase);
    float cos_2p = cos(phase * 2.0); float sin_2p = sin(phase * 2.0);

    float tmp_r01 = r01, tmp_i01 = i01;
    r01 = tmp_r01 * cos_p - tmp_i01 * sin_p; i01 = tmp_r01 * sin_p + tmp_i01 * cos_p;
    float tmp_r10 = r10, tmp_i10 = i10;
    r10 = tmp_r10 * cos_p - tmp_i10 * sin_p; i10 = tmp_r10 * sin_p + tmp_i10 * cos_p;
    float tmp_r11 = r11, tmp_i11 = i11;
    r11 = tmp_r11 * cos_2p - tmp_i11 * sin_2p; i11 = tmp_r11 * sin_2p + tmp_i11 * cos_2p;
}

void applyH0() {
    float inv_sqrt2 = 1.0 / sqrt(2.0);
    float o_r00 = r00, o_i00 = i00, o_r01 = r01, o_i01 = i01;
    float o_r10 = r10, o_i10 = i10, o_r11 = r11, o_i11 = i11;
    r00 = (o_r00 + o_r10) * inv_sqrt2; i00 = (o_i00 + o_i10) * inv_sqrt2;
    r10 = (o_r00 - o_r10) * inv_sqrt2; i10 = (o_i00 - o_i10) * inv_sqrt2;
    r01 = (o_r01 + o_r11) * inv_sqrt2; i01 = (o_i01 + o_i11) * inv_sqrt2;
    r11 = (o_r01 - o_r11) * inv_sqrt2; i11 = (o_i01 - o_i11) * inv_sqrt2;
    applyPhaseError(); 
}

void applyH1() {
    float inv_sqrt2 = 1.0 / sqrt(2.0);
    float o_r00 = r00, o_i00 = i00, o_r01 = r01, o_i01 = i01;
    float o_r10 = r10, o_i10 = i10, o_r11 = r11, o_i11 = i11;
    r00 = (o_r00 + o_r01) * inv_sqrt2; i00 = (o_i00 + o_i01) * inv_sqrt2;
    r01 = (o_r00 - o_r01) * inv_sqrt2; i01 = (o_i00 - o_i01) * inv_sqrt2;
    r10 = (o_r10 + o_r11) * inv_sqrt2; i10 = (o_i10 + o_i11) * inv_sqrt2;
    r11 = (o_r10 - o_r11) * inv_sqrt2; i11 = (o_i10 - o_i11) * inv_sqrt2;
    applyPhaseError(); 
}

void applyCNOT() {
    float tmp_r = r10; float tmp_i = i10;
    r10 = r11; i10 = i11;
    r11 = tmp_r; i11 = tmp_i;
    applyPhaseError(); 
}

void scanEnvironment() {
    canvas.fillScreen(BLACK);
    canvas.setCursor(10, 50);
    canvas.setTextColor(YELLOW);
    canvas.setTextSize(2);
    canvas.printf("Scanning Noise...");
    canvas.pushSprite(0, 0);
    
    wifi_count = WiFi.scanNetworks();
    base_noise = constrain(wifi_count * 0.05, 0.1, 0.8);
}

void updateNoiseDynamics() {
    // 3秒周期でノイズがなだらかに変動
    dynamic_noise = base_noise + sin(millis() / 3000.0) * 0.25;
    dynamic_noise = constrain(dynamic_noise, 0.0, 1.0);
    current_error = abs(dynamic_noise - calib_val);
    fidelity = 100.0 - (current_error * 100.0);
    if (fidelity < 0.0) fidelity = 0.0;

    // Fidelityが90%以上ならスコアが加算
    if (fidelity >= 90.0 && !calib_mode) {
        coherence_score += 1; 
    }
}

// シミュレータ画面の描画
void drawSimulator() {
    canvas.fillScreen(BLACK);
    canvas.setCursor(0, 0);
    
    // フィデリティ表示
    canvas.setTextSize(2);
    if (fidelity > 90) canvas.setTextColor(GREEN);
    else if (fidelity > 60) canvas.setTextColor(YELLOW);
    else canvas.setTextColor(RED);
    canvas.printf("Fidelity: %3.0f%%\n", fidelity);
    
    // スコア表示
    canvas.setTextColor(CYAN);
    canvas.printf("Score: %05lu\n", coherence_score);
    
    canvas.setTextColor(WHITE);
    canvas.setTextSize(1);
    canvas.printf(" Env Noise: %.2f Signals: %d\n", dynamic_noise, wifi_count);
    canvas.printf("--------------------------------\n");

    float probs[4] = {
        r00*r00 + i00*i00, r01*r01 + i01*i01, 
        r10*r10 + i10*i10, r11*r11 + i11*i11
    };
    const char* labels[4] = {"00", "01", "10", "11"};
    uint16_t colors[4] = {GREEN, BLUE, YELLOW, MAGENTA};

    canvas.setTextSize(2);
    for (int i = 0; i < 4; i++) {
        canvas.setTextColor(colors[i]);
        canvas.printf("[%s]:%3.0f%% ", labels[i], probs[i] * 100.0);
        int bar_w = probs[i] * 90;
        canvas.fillRect(145, 52 + (i * 20), bar_w, 12, colors[i]);
        canvas.println();
    }
    
    canvas.pushSprite(0, 0);
}

// キャリブレーション画面の描画
void drawCalibration() {
    canvas.fillScreen(BLACK);
    canvas.setCursor(0, 5);
    canvas.setTextColor(CYAN);
    canvas.setTextSize(2);
    canvas.printf("--- CALIBRATION ---\n");
    
    canvas.setTextColor(WHITE);
    canvas.setTextSize(1);
    canvas.printf("Target: Match [^] with Noise [*]\n\n");

    int target_x = dynamic_noise * 200;
    int user_x = calib_val * 200;
    
    canvas.drawLine(20, 60, 220, 60, DARKGREY);
    canvas.setTextColor(RED); canvas.setCursor(20 + target_x - 5, 45); canvas.print("*");
    canvas.setTextColor(GREEN); canvas.setCursor(20 + user_x - 5, 65); canvas.print("^");
    
    canvas.setCursor(0, 95);
    canvas.setTextColor(WHITE);
    canvas.setTextSize(2);
    canvas.printf("Score: %05lu\n", coherence_score);
    canvas.setTextSize(1);
    canvas.printf("Tuning: [%.2f] Err: %.2f\n", calib_val, current_error);
    canvas.printf("Use 'A'/'D' to Tune | 'M' to Return");
    
    canvas.pushSprite(0, 0);
}

void setup() {
    auto cfg = M5.config();
    M5Cardputer.begin(cfg, true);
    M5Cardputer.Display.setRotation(1);
    
    // スプライト（裏画面）のメモリ領域を確保
    canvas.createSprite(240, 135);
    
    scanEnvironment();
}

void loop() {
    M5Cardputer.update();
    updateNoiseDynamics();

    if (calib_mode) {
        if (M5Cardputer.Keyboard.isKeyPressed('a')) { calib_val -= 0.01; delay(20); }
        if (M5Cardputer.Keyboard.isKeyPressed('d')) { calib_val += 0.01; delay(20); }
        calib_val = constrain(calib_val, 0.0, 1.0);
        if (M5Cardputer.Keyboard.isKeyPressed('m')) { calib_mode = false; delay(200); }
        drawCalibration();
    } else {
        if (M5Cardputer.Keyboard.isKeyPressed('z')) { applyH0(); delay(200); }
        if (M5Cardputer.Keyboard.isKeyPressed('x')) { applyH1(); delay(200); }
        if (M5Cardputer.Keyboard.isKeyPressed('c')) { applyCNOT(); delay(200); }
        if (M5Cardputer.Keyboard.isKeyPressed('r')) {
            r00=1; i00=0; r01=0; i01=0; r10=0; i10=0; r11=0; i11=0; delay(200);
        }
        if (M5Cardputer.Keyboard.isKeyPressed('s')) { scanEnvironment(); delay(200); }
        if (M5Cardputer.Keyboard.isKeyPressed('m')) { calib_mode = true; delay(200); }
        drawSimulator();
    }
    delay(10);
}
```
